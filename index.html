Python Data Processing Example

This Python script demonstrates data cleaning and transformation using Pandas:

import pandas as pd
import numpy as np

# Load data
data = pd.read_csv('sales_data.csv')

# Clean and transform
data['date'] = pd.to_datetime(data['date'])
data['revenue'] = data['quantity'] * data['price']
data.fillna(0, inplace=True)

# Aggregate by month
monthly_sales = data.groupby(pd.Grouper(key='date', freq='M'))['revenue'].sum()

print(monthly_sales.head())

Security Hash Implementation

Secure password hashing using bcrypt in Python:

import bcrypt
import hashlib

def hash_password(password):
    # Generate salt
    salt = bcrypt.gensalt()
    # Hash password
    hashed = bcrypt.hashpw(password.encode(), salt)
    return hashed

def verify_password(password, hashed):
    return bcrypt.checkpw(password.encode(), hashed)

# Example usage
password = "SecurePass123!"
hashed = hash_password(password)
print(f"Stored hash: {hashed}")
print(f"Verification: {verify_password(password, hashed)}")

Data Visualization Example

Creating interactive charts with Plotly for data insights:

import plotly.express as px
import plotly.graph_objects as go

# Sample data
data = pd.DataFrame({
    'Month': ['Jan', 'Feb', 'Mar', 'Apr', 'May'],
    'Sales': [12000, 15000, 18000, 22000, 25000],
    'Expenses': [8000, 9000, 10000, 12000, 14000]
})

# Create interactive chart
fig = go.Figure()
fig.add_trace(go.Bar(x=data['Month'], y=data['Sales'], name='Sales'))
fig.add_trace(go.Bar(x=data['Month'], y=data['Expenses'], name='Expenses'))
fig.update_layout(title='Monthly Sales & Expenses', barmode='group')
fig.show()

REST API Implementation

Building a secure REST API with Node.js and Express:

const express = require('express');
const jwt = require('jsonwebtoken');
const app = express();

// Middleware
app.use(express.json());

// Authentication middleware
const authenticate = (req, res, next) => {
    const token = req.header('Authorization');
    if (!token) return res.status(401).send('Access denied');

    try {
        const verified = jwt.verify(token, process.env.JWT_SECRET);
        req.user = verified;
        next();
    } catch (err) {
        res.status(400).send('Invalid token');
    }
};

// Protected route
app.get('/api/users', authenticate, (req, res) => {
    res.json({ message: 'Protected data accessed' });
});
